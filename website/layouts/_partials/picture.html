{{- /* Responsive image from assets/images: WebP srcset + JPEG fallback. dict "src" "images/x.jpg" "alt" "…" "sizes" "…" "class" "…" "widths" (slice 480 800 1200 1600) "eager" false Theme pairs: when images/x-dark.jpg exists next to images/x.jpg, both are rendered (classes theme-img--light / theme-img--dark) and base.css shows the one matching the site theme (prefers-color-scheme, overridden by the toggle's data-theme on ). Both are loading="lazy", so the hidden one (display: none) is not downloaded. "theme" false turns this off. */ -}} {{- $dark := "" -}} {{- if and (ne .theme false) (not (strings.HasSuffix (path.BaseName .src) "-dark")) -}} {{- $ext := path.Ext .src -}} {{- $dark = printf "%s-dark%s" (strings.TrimSuffix $ext .src) $ext -}} {{- if not (resources.Get $dark) }}{{ $dark = "" }}{{ end -}} {{- end -}} {{- if $dark -}} {{- $cls := .class | default "" -}} {{- partial "picture-one.html" (merge . (dict "class" (printf "%s theme-img--light" $cls | strings.TrimSpace) "eager" false)) -}} {{- partial "picture-one.html" (merge . (dict "src" $dark "class" (printf "%s theme-img--dark" $cls | strings.TrimSpace) "eager" false)) -}} {{- else -}} {{- partial "picture-one.html" . -}} {{- end -}}
